cmake_minimum_required(VERSION 3.20)
project(fltube LANGUAGES CXX)
find_package(FLTK 1.4.5 EXACT CONFIG REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(CURL REQUIRED IMPORTED_TARGET libcurl)
pkg_check_modules(FONTCONFIG REQUIRED IMPORTED_TARGET fontconfig)
find_package(Threads REQUIRED)
find_package(nlohmann_json REQUIRED)
file(STRINGS VERSION FLTUBE_VERSION LIMIT_COUNT 1)
add_executable(fltube src/fltube_utils.cxx src/gnugettext_utils.cxx src/FLTube_View.cxx
    src/FLTube.cxx src/configuration_manager.cxx src/userdata_manager.cxx
    src/ytdlp_helper.cxx src/cache.cxx src/custom_widgets.cxx)
target_compile_features(fltube PRIVATE cxx_std_17)
target_compile_definitions(fltube PRIVATE VERSION_STRING="${FLTUBE_VERSION}")
target_include_directories(fltube PRIVATE include integration)
include(integration/desktop-menu.cmake)
de_xr_desktop_menu(fltube)
target_link_libraries(fltube PRIVATE fltk::fltk-shared fltk::images-shared PkgConfig::CURL PkgConfig::FONTCONFIG Threads::Threads nlohmann_json::nlohmann_json)
include(GNUInstallDirs)
# Resource paths follow the selected installation prefix.
target_compile_definitions(fltube PRIVATE FLTUBE_PREFIX="${CMAKE_INSTALL_PREFIX}")
set_target_properties(fltube PROPERTIES INSTALL_RPATH "$<TARGET_FILE_DIR:fltk::fltk-shared>")
install(TARGETS fltube RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES fltube.conf DESTINATION etc/fltube)
install(DIRECTORY resources/img DESTINATION ${CMAKE_INSTALL_DATADIR}/fltube/resources)
install(DIRECTORY resources/icons/ DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/fltube)
configure_file(resources/desktop/fltube.desktop.in fltube.desktop @ONLY)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/fltube.desktop" DESTINATION ${CMAKE_INSTALL_DATADIR}/applications RENAME org.domed.FLTube.desktop)
find_program(MSGFMT msgfmt REQUIRED)
foreach(lang en es pt_BR)
  set(mo "${CMAKE_CURRENT_BINARY_DIR}/${lang}/FLTube.mo")
  add_custom_command(OUTPUT "${mo}" COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_CURRENT_BINARY_DIR}/${lang}"
    COMMAND ${MSGFMT} -o "${mo}" "${CMAKE_CURRENT_SOURCE_DIR}/locales/${lang}/LC_MESSAGES/FLTube.po"
    DEPENDS "locales/${lang}/LC_MESSAGES/FLTube.po")
  add_custom_target(locale-${lang} ALL DEPENDS "${mo}")
  install(FILES "${mo}" DESTINATION "${CMAKE_INSTALL_DATADIR}/locale/${lang}/LC_MESSAGES")
endforeach()
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_executable(playback-driver tests/playback_driver.cpp)
  target_compile_features(playback-driver PRIVATE cxx_std_17)
  target_include_directories(playback-driver PRIVATE include)
  add_test(NAME playback-launch COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_playback.py $<TARGET_FILE:playback-driver>)
endif()
